// Fungsi pembersih nama file agar aman dari karakter '#' atau spasi function sanitizeStorageFileName(name) { return name .replace(/\s+/g, '_') // Ubah spasi jadi underscore .replace(/[^a-zA-Z0-9._-]/g, ''); // Hapus '#' dan simbol khusus lainnya } // 1. TAMBAH LAGU BARU async function simpanLagu() { const title = document.getElementById('title').value.trim(); const genre = document.getElementById('genre').value.trim(); const lyrics = document.getElementById('lyrics').value.trim(); const coverFileInput = document.getElementById('cover_file'); const audioFileInput = document.getElementById('audio_file'); const karaokeFileInput = document.getElementById('karaoke_file'); const statusDiv = document.getElementById('status'); const btnSubmit = document.getElementById('btnSubmit'); const coverFile = coverFileInput.files[0]; const audioFile = audioFileInput.files[0]; const karaokeFile = karaokeFileInput.files[0] || null; if (!title || !genre || !coverFile || !audioFile) { statusDiv.style.color = '#ef4444'; statusDiv.innerText = '❌ Judul, Genre, Gambar Cover, dan File MP3 Utama wajib diisi!'; return; } try { btnSubmit.disabled = true; btnSubmit.innerText = '⏳ Mengunggah file...'; statusDiv.style.color = '#f59e0b'; statusDiv.innerText = '⏳ Sedang mengunggah berkas ke Supabase...'; const timestamp = Date.now(); // Upload Cover const coverPath = `covers/${timestamp}_${sanitizeStorageFileName(coverFile.name)}`; const { error: coverError } = await _supabase.storage.from('lagu-suno').upload(coverPath, coverFile, { cacheControl: '3600', upsert: true }); if (coverError) throw new Error('Gagal upload cover: ' + coverError.message); // Upload Audio Utama const audioPath = `audio/${timestamp}_${sanitizeStorageFileName(audioFile.name)}`; const { error: audioError } = await _supabase.storage.from('lagu-suno').upload(audioPath, audioFile, { cacheControl: '3600', upsert: true }); if (audioError) throw new Error('Gagal upload audio utama: ' + audioError.message); // Upload Audio Karaoke let karaokeUrl = null; if (karaokeFile) { statusDiv.innerText = '⏳ Sedang mengunggah Audio Karaoke...'; const karaokePath = `karaoke/${timestamp}_${sanitizeStorageFileName(karaokeFile.name)}`; const { error: kError } = await _supabase.storage.from('lagu-suno').upload(karaokePath, karaokeFile, { cacheControl: '3600', upsert: true }); if (kError) throw new Error('Gagal upload karaoke: ' + kError.message); const { data: kData } = _supabase.storage.from('lagu-suno').getPublicUrl(karaokePath); karaokeUrl = kData.publicUrl; } const { data: coverUrlData } = _supabase.storage.from('lagu-suno').getPublicUrl(coverPath); const { data: audioUrlData } = _supabase.storage.from('lagu-suno').getPublicUrl(audioPath); statusDiv.innerText = '⏳ Menyimpan data ke database...'; const { error: dbError } = await _supabase .from('songs') .insert([{ title, genre, lyrics, cover_url: coverUrlData.publicUrl, audio_url: audioUrlData.publicUrl, karaoke_url: karaokeUrl }]); if (dbError) throw dbError; statusDiv.style.color = '#22c55e'; statusDiv.innerText = '✅ Berhasil! Lagu & file karaoke ter-upload.'; document.getElementById('title').value = ''; document.getElementById('genre').value = ''; document.getElementById('lyrics').value = ''; coverFileInput.value = ''; audioFileInput.value = ''; karaokeFileInput.value = ''; fetchAdminSongs(); } catch (err) { statusDiv.style.color = '#ef4444'; statusDiv.innerText = '❌ ' + err.message; } finally { btnSubmit.disabled = false; btnSubmit.innerText = 'Simpan & Unggah Lagu'; } } // 4. UPDATE DATA LAGU (EDIT) async function updateLagu() { const id = document.getElementById('edit_id').value; const title = document.getElementById('edit_title').value.trim(); const genre = document.getElementById('edit_genre').value.trim(); const lyrics = document.getElementById('edit_lyrics').value.trim(); const coverFileInput = document.getElementById('edit_cover_file'); const audioFileInput = document.getElementById('edit_audio_file'); const karaokeFileInput = document.getElementById('edit_karaoke_file'); let cover_url = document.getElementById('edit_old_cover').value || null; let audio_url = document.getElementById('edit_old_audio').value || null; let karaoke_url = document.getElementById('edit_old_karaoke').value || null; const editStatus = document.getElementById('editStatus'); const btnUpdate = document.getElementById('btnUpdateSubmit'); if (!title || !genre) { editStatus.style.color = '#ef4444'; editStatus.innerText = '❌ Judul dan Genre wajib diisi!'; return; } try { btnUpdate.disabled = true; btnUpdate.innerText = '⏳ Memproses...'; editStatus.style.color = '#f59e0b'; editStatus.innerText = '⏳ Menyimpan perubahan...'; const timestamp = Date.now(); // Ganti Cover jika dipilih if (coverFileInput.files.length > 0) { const coverFile = coverFileInput.files[0]; const coverPath = `covers/${timestamp}_${sanitizeStorageFileName(coverFile.name)}`; const { error: coverError } = await _supabase.storage.from('lagu-suno').upload(coverPath, coverFile, { cacheControl: '3600', upsert: true }); if (coverError) throw new Error('Gagal upload cover baru: ' + coverError.message); const { data: coverUrlData } = _supabase.storage.from('lagu-suno').getPublicUrl(coverPath); cover_url = coverUrlData.publicUrl; } // Ganti Audio Utama jika dipilih if (audioFileInput.files.length > 0) { const audioFile = audioFileInput.files[0]; const audioPath = `audio/${timestamp}_${sanitizeStorageFileName(audioFile.name)}`; const { error: audioError } = await _supabase.storage.from('lagu-suno').upload(audioPath, audioFile, { cacheControl: '3600', upsert: true }); if (audioError) throw new Error('Gagal upload audio baru: ' + audioError.message); const { data: audioUrlData } = _supabase.storage.from('lagu-suno').getPublicUrl(audioPath); audio_url = audioUrlData.publicUrl; } // Ganti/Upload Audio Karaoke jika dipilih if (karaokeFileInput.files.length > 0) { const karaokeFile = karaokeFileInput.files[0]; const karaokePath = `karaoke/${timestamp}_${sanitizeStorageFileName(karaokeFile.name)}`; const { error: kError } = await _supabase.storage.from('lagu-suno').upload(karaokePath, karaokeFile, { cacheControl: '3600', upsert: true }); if (kError) throw new Error('Gagal upload file karaoke: ' + kError.message); const { data: kData } = _supabase.storage.from('lagu-suno').getPublicUrl(karaokePath); karaoke_url = kData.publicUrl; } // Update ke Tabel Supabase const { error: updateError } = await _supabase .from('songs') .update({ title, genre, lyrics, cover_url, audio_url, karaoke_url: karaoke_url ? karaoke_url.trim() : null }) .eq('id', id); if (updateError) throw updateError; editStatus.style.color = '#22c55e'; editStatus.innerText = '✅ Berhasil! Data & Karaoke tersimpan.'; setTimeout(() => { closeEditModal(); fetchAdminSongs(); }, 1000); } catch (err) { editStatus.style.color = '#ef4444'; editStatus.innerText = '❌ ' + err.message; } finally { btnUpdate.disabled = false; btnUpdate.innerText = 'Simpan Perubahan'; } }
