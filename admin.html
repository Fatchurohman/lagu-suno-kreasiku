OPEN & CLOSE EDIT MODAL function openEditModal(id) { const song = songsCache.find(s => String(s.id) === String(id)); if (!song) return; document.getElementById('edit_id').value = song.id; document.getElementById('edit_title').value = song.title || ''; document.getElementById('edit_genre').value = song.genre || ''; document.getElementById('edit_lyrics').value = song.lyrics || ''; // Pastikan URL lama tidak undefined / string "null" document.getElementById('edit_old_cover').value = song.cover_url || ''; document.getElementById('edit_old_audio').value = song.audio_url || ''; document.getElementById('edit_old_karaoke').value = (song.karaoke_url && song.karaoke_url !== 'null') ? song.karaoke_url : ''; document.getElementById('edit_cover_file').value = ''; document.getElementById('edit_audio_file').value = ''; document.getElementById('edit_karaoke_file').value = ''; const editStatus = document.getElementById('editStatus'); editStatus.innerText = ''; editStatus.style.color = '#fff'; document.getElementById('editModal').classList.add('show'); } // 4. UPDATE DATA LAGU (EDIT) async function updateLagu() { const id = document.getElementById('edit_id').value; const title = document.getElementById('edit_title').value.trim(); const genre = document.getElementById('edit_genre').value.trim(); const lyrics = document.getElementById('edit_lyrics').value.trim(); const coverFileInput = document.getElementById('edit_cover_file'); const audioFileInput = document.getElementById('edit_audio_file'); const karaokeFileInput = document.getElementById('edit_karaoke_file'); let cover_url = document.getElementById('edit_old_cover').value.trim() || null; let audio_url = document.getElementById('edit_old_audio').value.trim() || null; let karaoke_url = document.getElementById('edit_old_karaoke').value.trim() || null; const editStatus = document.getElementById('editStatus'); const btnUpdate = document.getElementById('btnUpdateSubmit'); if (!title || !genre) { editStatus.style.color = '#ef4444'; editStatus.innerText = '❌ Judul dan Genre wajib diisi!'; return; } try { btnUpdate.disabled = true; btnUpdate.innerText = '⏳ Memproses...'; editStatus.style.color = '#f59e0b'; editStatus.innerText = '⏳ Memeriksa berkas...'; const timestamp = Date.now(); // 1. Upload Cover Baru jika dipilih if (coverFileInput.files && coverFileInput.files.length > 0) { editStatus.innerText = '⏳ Mengunggah Cover baru...'; const coverFile = coverFileInput.files[0]; const coverPath = `covers/${timestamp}_${sanitizeStorageFileName(coverFile.name)}`; const { error: coverError } = await _supabase.storage.from('lagu-suno').upload(coverPath, coverFile, { cacheControl: '3600', upsert: true }); if (coverError) throw new Error('Gagal upload cover: ' + coverError.message); const { data: coverUrlData } = _supabase.storage.from('lagu-suno').getPublicUrl(coverPath); cover_url = coverUrlData.publicUrl; } // 2. Upload Audio Utama Baru jika dipilih if (audioFileInput.files && audioFileInput.files.length > 0) { editStatus.innerText = '⏳ Mengunggah Audio Utama baru...'; const audioFile = audioFileInput.files[0]; const audioPath = `audio/${timestamp}_${sanitizeStorageFileName(audioFile.name)}`; const { error: audioError } = await _supabase.storage.from('lagu-suno').upload(audioPath, audioFile, { cacheControl: '3600', upsert: true }); if (audioError) throw new Error('Gagal upload audio: ' + audioError.message); const { data: audioUrlData } = _supabase.storage.from('lagu-suno').getPublicUrl(audioPath); audio_url = audioUrlData.publicUrl; } // 3. Upload Audio Karaoke Baru jika dipilih if (karaokeFileInput.files && karaokeFileInput.files.length > 0) { editStatus.innerText = '⏳ Mengunggah Audio Karaoke...'; const karaokeFile = karaokeFileInput.files[0]; const karaokePath = `karaoke/${timestamp}_${sanitizeStorageFileName(karaokeFile.name)}`; const { error: kError } = await _supabase.storage.from('lagu-suno').upload(karaokePath, karaokeFile, { cacheControl: '3600', upsert: true }); if (kError) throw new Error('Gagal upload MP3 Karaoke: ' + kError.message); const { data: kData } = _supabase.storage.from('lagu-suno').getPublicUrl(karaokePath); karaoke_url = kData.publicUrl; } editStatus.innerText = '⏳ Menyimpan perubahan ke database...'; // 4. Update Database const { error: updateError } = await _supabase .from('songs') .update({ title: title, genre: genre, lyrics: lyrics, cover_url: cover_url, audio_url: audio_url, karaoke_url: (karaoke_url && karaoke_url !== 'null') ? karaoke_url : null }) .eq('id', id); if (updateError) throw new Error('Database Error: ' + updateError.message); editStatus.style.color = '#22c55e'; editStatus.innerText = '✅ Berhasil! File Karaoke tersimpan.'; // Refresh data lagu & tutup modal await fetchAdminSongs(); setTimeout(() => { closeEditModal(); }, 1200); } catch (err) { console.error('Update Lagu Error:', err); editStatus.style.color = '#ef4444'; editStatus.innerText = '❌ ' + err.message; } finally { btnUpdate.disabled = false; btnUpdate.innerText = 'Simpan Perubahan'; } }
